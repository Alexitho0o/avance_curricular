000624: def _trimester_level_to_semester(values: pd.Series) -> pd.Series:
000625:     """Convierte nivel administrativo trimestral a semestre curricular MU.
000626: 
000627:     Regla institucional inmutable:
000628:     1→1, 2→2, 3→2, 4→3, 5→4, 6→4, 7→5, 8→6, 9→6, 10→7, 11→8, 12→8.
000629:     Fallback para niveles >12: ceil(n*2/3), equivalente entero ((2n+2)//3).
000630:     """
000631:     nivel = pd.to_numeric(values, errors="coerce").round()
000632:     map_eq = {
000633:         1: 1, 2: 2, 3: 2,
000634:         4: 3, 5: 4, 6: 4,
000635:         7: 5, 8: 6, 9: 6,
000636:         10: 7, 11: 8, 12: 8,
000637:     }
000638:     sem = nivel.map(map_eq)
000639:     fallback_mask = sem.isna() & nivel.notna() & nivel.gt(12)
000640:     sem.loc[fallback_mask] = ((nivel.loc[fallback_mask] * 2 + 2) // 3)
